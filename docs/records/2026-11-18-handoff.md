## 5. 驗證方式（本輪用的）

無頭截圖用 Python Playwright（已安裝），登入用臨時 session：

```bash
docker compose exec -T redis redis-cli set session:claudetest_war 1 EX 900   # uid 1 = wei
```

```python
# 存成任意 .py，參數是輸出資料夾
import asyncio, sys
from playwright.async_api import async_playwright
S = sys.argv[1]
async def main():
    async with async_playwright() as p:
        b = await p.chromium.launch(headless=True, args=["--use-gl=angle", "--use-angle=metal"])
        for name, w, h, mob in [("desk", 1920, 1080, False), ("phone", 390, 844, True)]:
            c = await b.new_context(viewport={"width": w, "height": h}, is_mobile=mob, has_touch=mob)
            await c.add_cookies([{"name": "qiji_session", "value": "claudetest_war", "url": "http://127.0.0.1:8750"}])
            pg = await c.new_page(); errs = []
            pg.on("console", lambda m: errs.append(m.text) if m.type == "error" else None)
            pg.on("pageerror", lambda e: errs.append(str(e)))
            await pg.goto("http://127.0.0.1:8750/war?id=57"); await pg.wait_for_timeout(15000)
            await pg.screenshot(path=f"{S}/{name}.png", timeout=90000)
            print(name, "errors:", errs[:5])
        await b.close()
asyncio.run(main())
```

- 測完 `redis-cli del session:claudetest_war`。
